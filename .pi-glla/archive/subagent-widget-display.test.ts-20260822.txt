const NOW = Date.parse("2026-08-17T20:10:00Z");
const MANAGER_KEY = Symbol.for("pi-subagents:manager");

function installMockManager(
  getRecord: (id: string) => any | undefined,
  listAgents?: () => any[],
): void {
  (globalThis as any)[MANAGER_KEY] = {
    getRecord,
    manager: listAgents ? { listAgents } : undefined,
  };
}

function uninstallManager(): void {
  delete (globalThis as any)[MANAGER_KEY];
}

function createMockExtensionContext(overrides: any = {}): any {
  return {
    ui: {
      notify: () => {
        /* no-op */
      },
      select: async (_prompt: string, options: string[]) => options[0] ?? null,
      custom: async () => {
        /* no-op */
      },
    },
    cwd: tmpCwd(),
    hasUI: true,
    model: { provider: "test", id: "test-model" },
    ...overrides,
  };
}

const pi = new MockPi();
activate(pi.api);

test("getSubagentActivitySnapshot returns correct shape", () => {
  __testOnlyClearSubagentHangProbes();

  // Add a mock probe via the public test API
  upsertSubagentHangProbe(
    "abc12345-def678",
    "Auditor",
    "reading source…",
    NOW - 5000,
  );

  const snapshot = getSubagentActivitySnapshot(NOW);

  assert.equal(snapshot.length, 1);
  const entry = snapshot[0]!;
  assert.equal(entry.recordId, "abc12345-def678");
  assert.equal(entry.agentType, "Auditor");
  assert.equal(entry.summary, "reading source…");
  assert.equal(entry.spawnedAt, NOW - 5000);
  // silentMs = NOW - lastProgressAt = NOW - (NOW - 5000) = 5000
  assert.equal(entry.silentMs, 5000);
  assert.equal(entry.endedAt, undefined);
});

test("getSubagentActivitySnapshot returns empty array when no probes", () => {
  __testOnlyClearSubagentHangProbes();

  const snapshot = getSubagentActivitySnapshot(NOW);
  assert.deepEqual(snapshot, []);
});

test("getSubagentActivitySnapshot includes ended subagents", () => {
  __testOnlyClearSubagentHangProbes();

  upsertSubagentHangProbe("live-123", "Designer", "designing…", NOW - 10000);
  upsertSubagentHangProbe(
    "ended-456",
    "Explore",
    "completed task",
    NOW - 20000,
  );
  endSubagentHangProbe("ended-456", NOW - 1000);

  const snapshot = getSubagentActivitySnapshot(NOW);

  // Snapshot returns ALL probes (widget filters ended ones)
  assert.equal(snapshot.length, 2);
  const liveEntry = snapshot.find((e) => e.recordId === "live-123")!;
  const endedEntry = snapshot.find((e) => e.recordId === "ended-456")!;
  assert.equal(liveEntry.endedAt, undefined);
  assert.equal(endedEntry.endedAt, NOW - 1000);
});

test("buildWidgetLines renders subagentActivity in widget", () => {
  const state: State = {
    goal: null,
    list: [],
  };

  const subagentActivity = [
    {
      recordId: "abc12345-def678",
      agentType: "Auditor",
      summary: "reading source…",
      spawnedAt: NOW - 10000,
      silentMs: 150000, // 2m 30s
      endedAt: undefined,
    },
    {
      recordId: "xyz98765-uvw432",
      agentType: "Designer",
      summary: "reviewing PR #42",
      spawnedAt: NOW - 50000,
      silentMs: 60000, // 1m
      endedAt: undefined,
    },
  ];

  const lines = buildWidgetLines(state, null, NOW, undefined, 120, {
    subagentActivity,
  })!;

  const rendered = lines.join("\n");
  // Check for the subagent indicator (⟳) and truncated record ID
  assert.match(
    rendered,
    /⟳ Auditor \[abc12345\]: reading source… · 2m 30s ago/,
  );
  assert.match(
    rendered,
    /⟳ Designer \[xyz98765\]: reviewing PR #42 · 1m 0s ago/,
  );
});

test("buildWidgetLines filters ended subagents", () => {
  const state: State = { goal: null, list: [] };

  const subagentActivity = [
    {
      recordId: "live-111",
      agentType: "Auditor",
      summary: "working…",
      spawnedAt: NOW - 10000,
      silentMs: 5000,
      endedAt: undefined,
    },
    {
      recordId: "ended-222",
      agentType: "Designer",
      summary: "done",
      spawnedAt: NOW - 20000,
      silentMs: 600000, // 10m old
      endedAt: NOW - 1000,
    },
  ];

  const lines = buildWidgetLines(state, null, NOW, undefined, 120, {
    subagentActivity,
  })!;

  const rendered = lines.join("\n");
  // Live subagent should appear
  assert.match(rendered, /⟳ Auditor \[live-111\]: working…/);
  // Ended subagent should NOT appear (filter: endedAt === undefined && silentMs < 10min)
  assert.doesNotMatch(rendered, /ended-222/);
});

test("buildWidgetLines filters stale subagents (>10min silent)", () => {
  const state: State = { goal: null, list: [] };

  const subagentActivity = [
    {
      recordId: "fresh-111",
      agentType: "Auditor",
      summary: "working…",
      spawnedAt: NOW - 10000,
      silentMs: 5000, // 5s - fresh
      endedAt: undefined,
    },
    {
      recordId: "stale-222",
      agentType: "Designer",
      summary: "stuck…",
      spawnedAt: NOW - 20000,
      silentMs: 11 * 60_000, // 11min - stale
      endedAt: undefined,
    },
  ];

  const lines = buildWidgetLines(state, null, NOW, undefined, 120, {
    subagentActivity,
  })!;

  const rendered = lines.join("\n");
  // Fresh subagent should appear
  assert.match(rendered, /⟳ Auditor \[fresh-111\]: working…/);
  // Stale subagent should NOT appear
  assert.doesNotMatch(rendered, /stale-222/);
});

  uninstallManager(); // Ensure no manager

});
